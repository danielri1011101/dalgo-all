\begin{frame} {Dynamic programming: Top-Down}
	\[
		\begin{array}{l}
			\mbox{Items } \mathbf{x}= x_1, \ldots, x_n\\
			x_i= (w_i, v_i)\\
			\mbox{Capacity } W \\
			 \\
			  
		\end{array}
	\]\pause
	Let $V$ denote the max-value function that solves Knapsack.
\end{frame}
%
%nextframe
\begin{frame}
	We have the recurrence
	\[
		V(n,W)= \max
		\left\{
		\begin{array}{l} 
			v_n + V(n-1, W-w_n)\\
			V(n-1, W)
		\end{array} 
		\right.
	\]\pause
	\begin{itemize}
		\item \emph{Recursion + $V$-table!}
	\end{itemize}
\end{frame}
