	\[
		V(n,W)= \max
		\left\{
		\begin{array}{l} 
			v_n + V(n-1, W-w_n)\\
			V(n-1, W)
		\end{array} 
		\right.
	\]\pause
	\begin{itemize}
		\item \emph{Recursion + $V$-table!}
	\end{itemize}

\begin{frame}
	\textbf{Problem:} Compute \( \binom{n}{k} \).
	\[
		\begin{array}{l}
			P\equiv 0 \leq k \leq n \ \ \land \ \texttt{*out} = \perp\\
			Q\equiv \texttt{*out} = \binom{n}{k}
		\end{array}
	\]
	\textbf{Solution:} A program $S$ such that
	\begin{enumerate}
		\item $P \equiv \textbf{true}$
		\item Run $S$
		\item $Q \equiv \textbf{true}$
	\end{enumerate}
	
\end{frame}
%
%nextframe
\begin{frame}
	\textbf{Problem:} Knapsack with capacity $W$ and items
	\( (w_1, v_1), \ldots, (w_n, v_n) \)
	\[
		\begin{array}{cc}
			P\equiv & \forall i: w_i, v_i > 0 \ \land \ W > 0
				\land \ \texttt{*out} = \perp\\
			Q\equiv & \texttt{*out} =
				\max \{\sum_{i=1}^n x_i\cdot v_i
					\ | \ \sum_{i=1}^n x_i\cdot w_i
				 			  \leq W \ | \\ 
			     & x_i = 0 \lor x_i = 1 \ \ \}
		\end{array}
	\]
	\textbf{Solution:} A program $S$ such that
	\begin{enumerate}
		\item $P \equiv \textbf{true}$
		\item Run $S$
		\item $Q \equiv \textbf{true}$
	\end{enumerate}
	
\end{frame}
%
%nextframe
\begin{frame}
	The sequence
	\begin{enumerate}
		\item $P \equiv \textbf{true}$
		\item Run $S$
		\item $Q \equiv \textbf{true}$
	\end{enumerate}
	is the \emph{specification} of $S$. $P$ is the \emph{precondition}, and
	$Q$ is the \emph{postcondition}.
	\[
		\mbox{Notation:}\ P \{S\} Q.
	\]
\end{frame}

\begin{frame}{Graphs-Theory}
	A \emph{graph} $G$ is the data $(N, A)$ where
	\begin{itemize}
		\item $N$ is a finitie set of elements called \emph{nodes}.
		\item $A \subset \binom{N}{2}$ is the set of \emph{edges}
	\end{itemize}
	where 
	\[
		\binom{N}{2}=
		\left\{
			S \subset N \mid \# S = 2
		\right\}
	\]
\end{frame}
