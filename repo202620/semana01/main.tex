\begin{frame}{}
	\begin{theorem}
		If $A$ is a square matrix with non-zero diagonal entries, there
		exist a lower-triangular matrix $L$ and an upper-triangular matrix
		$U$ such that
		\[
			LA = U.
		\]
		Then
		\[
			\mathrm{det}(A) = \mathrm{det}(U)/\mathrm{det}(L).
		\]
	\end{theorem}
\end{frame}
