\begin{frame}{Directed Graphs - Theory}
	A \emph{directed graph} $G = (N, A)$ is given by
	\begin{itemize}
		\item $N$ is a finitie set of \emph{nodes}.
		\item $A \subset N \times N - \Delta_N$

			is the set of \emph{arrows}
	\end{itemize}
	where 
	\[
		\Delta_N=
		\left\{
			(u, u) \vert u \in N
		\right\}
	\]
\end{frame}
