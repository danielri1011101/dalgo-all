\begin{frame}{Graphs-Theory}
	A \emph{graph} $G$ is the data $(N, A)$ where
	\begin{itemize}
		\item $N$ is a finitie set of elements called \emph{nodes}.
		\item $A \subset \binom{N}{2}$ is the set of \emph{edges}
	\end{itemize}
	where 
	\[
		\binom{N}{2}=
		\left\{
			S \subset N \mid \# S = 2
		\right\}
	\]
\end{frame}
